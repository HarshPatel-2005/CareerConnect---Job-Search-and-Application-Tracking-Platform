    const express = require('express');
    const db = require('../../config/db');
    const { authenticateToken } = require('../middleware/authMiddleware');
    const router = express.Router();
    const statuses = ['applied', 'under_review', 'interview', 'offered', 'rejected'];
    const metadata = `a.id, a.job_id, a.user_id, a.resume_name, a.status,
        UNIX_TIMESTAMP(a.applied_at) * 1000 AS applied_at`;
    const publicRecord = row => ({ ...row, applied_at: new Date(Number(row.applied_at)).toISOString() });
    router.use(authenticateToken);
    router.get('/', async (req, res, next) => {
        if (req.user.role !== 'job_seeker') return res.status(403).json({ error: 'Job seeker account required.' });
        try {
            const [rows] = await db.query(`SELECT ${metadata}, j.title, j.location, c.name AS company_name
                FROM applications a JOIN jobs j ON j.id = a.job_id JOIN companies c ON c.id = j.company_id
                WHERE a.user_id = ? ORDER BY a.applied_at DESC, a.id DESC`, [req.user.id]);
            res.json({ applications: rows.map(publicRecord) });
        } catch (error) { next(error); }
    });
    // Recruiters may only view applicants to their own company's postings.
    router.get('/job/:id', async (req, res, next) => {
        if (req.user.role !== 'recruiter' || !req.user.company_id) return res.status(403).json({ error: 'Recruiter account required.' });
        try {
            const [jobs] = await db.query('SELECT id, title FROM jobs WHERE id = ? AND company_id = ?', [req.params.id, req.user.company_id]);
            if (!jobs.length) return res.status(404).json({ error: 'Job not found.' });
            const [rows] = await db.query(`SELECT ${metadata}, u.full_name, u.email
                FROM applications a JOIN users u ON u.id = a.user_id
                WHERE a.job_id = ? ORDER BY a.applied_at DESC, a.id DESC`, [req.params.id]);
            res.json({ job: jobs[0], applications: rows.map(publicRecord) });
        } catch (error) { next(error); }
    });
    router.get('/:id/resume', async (req, res, next) => {
        try {
            const [rows] = await db.query(`SELECT a.user_id, j.company_id, a.resume_name, a.resume_mime_type, a.resume_content
                FROM applications a JOIN jobs j ON j.id = a.job_id WHERE a.id = ?`, [req.params.id]);
            const row = rows[0];
            const allowed = row && ((req.user.role === 'job_seeker' && String(row.user_id) === String(req.user.id))
                || (req.user.role === 'recruiter' && String(row.company_id) === String(req.user.company_id)));
            if (!allowed) return res.status(404).json({ error: 'Application not found.' });
            res.attachment(row.resume_name).type(row.resume_mime_type).send(row.resume_content);
        } catch (error) { next(error); }
    });
    router.patch('/:id/status', async (req, res, next) => {
        if (req.user.role !== 'recruiter' || !req.user.company_id) return res.status(403).json({ error: 'Recruiter account required.' });
        if (!statuses.includes(req.body?.status)) return res.status(400).json({ error: 'Invalid application status.' });
        let connection;
        try {
            connection = await db.getConnection();
            await connection.beginTransaction();
            const [applications] = await connection.query(`SELECT a.job_id FROM applications a
                JOIN jobs j ON j.id = a.job_id WHERE a.id = ? AND j.company_id = ?`, [req.params.id, req.user.company_id]);
            if (!applications.length) { await connection.rollback(); return res.status(404).json({ error: 'Application not found.' }); }
            const jobId = applications[0].job_id;
            await connection.query('SELECT id FROM jobs WHERE id = ? FOR UPDATE', [jobId]);
            await connection.query('UPDATE applications SET status = ? WHERE id = ?', [req.body.status, req.params.id]);
            // Recompute from source records, preserving consistency when revisiting a status.
            const [counts] = await connection.query(`SELECT COUNT(*) AS total,
                SUM(status = 'applied') AS awaiting, SUM(status IN ('under_review', 'rejected')) AS reviewed,
                SUM(status = 'interview') AS contacting, SUM(status = 'offered') AS hired
                FROM applications WHERE job_id = ?`, [jobId]);
            const c = counts[0];
            await connection.query(`UPDATE jobs SET candidates_count = ?, awaiting_count = ?, reviewed_count = ?,
                contacting_count = ?, hired_count = ? WHERE id = ?`, [c.total, c.awaiting || 0, c.reviewed || 0, c.contacting || 0, c.hired || 0, jobId]);
            await connection.commit();
            res.json({ status: req.body.status });
        } catch (error) { if (connection) await connection.rollback(); next(error); }
        finally { if (connection) connection.release(); }
    });
    router.use((error, _req, res, _next) => {
        console.error('Application request failed:', error);
        res.status(500).json({ error: 'Unable to load or update applications right now.' });
    });
    module.exports = router;
